function CStnSpecificGUI()
{
    var blnIsConsole;
    try
    {
        blnIsConsole = window.external.status.IsConsoleStation;
    }
    catch(e)
    {
        blnIsConsole = false;
    }

    if (true == blnIsConsole)
    {
        CurrentState = window.external.status.FieldState(5);

        if(CurrentState == nServerUnavailable)
        {

            if(EditorType == nCombinedScriptEditor)
            {
                if(cboScriptType.selectedIndex != -1)
                {
                    cboScriptType.selectedIndex = -1;
                }
            }

            try
            {
                // Don't worry about security here
                CombinedEd.SaveWorking();
            }
            catch(e)
            {
                cmdSave.style.visibility   = "hidden";
                cmdUndo.style.visibility   = "hidden";
                cmdCommit.style.visibility = "hidden";
                cmdReset.style.visibility  = "hidden";
                txtCannotSave.style.visibility = "visible";
                return true;
            }
            ReportMessage(GetLocalI18NString(419));
            CombinedEd.UnsetView();
            return true;
        }
    }
    return false;
}

function IsSecure()
{
    try
    {
        var SecLevel = window.external.SecurityLevel;
        if(null == SecLevel)
        {
            // Assume not secure if no Station model
            return false;
        }
        else if(SecLevel >= 4)
        {
            return true;
        }
        else
        {
            return false;
        }
    }
    catch(e)
    {
        // Assume not secure if errors encountered
        return false;
    }
}

IsConsoleStationFetched = false;
bIsConsoleStation = false;	// This is default

//
// Returns true if we are running from a console station
// that is not connected to the server.
//
function IsDisconnectedCStn()
{
    var blnIsConsole = bIsConsoleStation;

    try
    {
		if (IsConsoleStationFetched == false)
		{
	        blnIsConsole = window.external.status.IsConsoleStation;
			IsConsoleStationFetched = true;
			bIsConsoleStation = blnIsConsole;
		}
    }
    catch(e)
    {
		IsConsoleStationFetched = true;
        blnIsConsole = bIsConsoleStation;	// Use default
    }

    if (true == blnIsConsole)
    {
        CurrentState = window.external.status.FieldState(5);

        if (CurrentState == nServerUnavailable)
        {
            return true;
        }
    }

    return false;
}

IsDisconnectedCStn();

function CombinedEd_OnChangeState()
{
    try
    {
        var blnSave     = false;
        var blnCommit   = false;
        var blnRestore  = false;
        var blnUndo     = false;
    
        switch(CombinedEd.State)
        {
        case -1:
        case 0:
            // Leave everything disabled
            break;
        case 1:
            blnSave = true
            blnUndo = true
            break;
        case 2:
            blnRestore = true
            blnCommit = true
            break;
        case 3:
            blnSave = true
            blnRestore = true
            blnCommit = true
            blnUndo = true
            break;
        default:
            // Leave everything disabled
            break;
        }
    
        cmdSave.disabled = !blnSave;
        cmdCommit.disabled = !blnCommit;
        cmdReset.disabled = !blnRestore;
        cmdUndo.disabled = !blnUndo;
    }
    catch(e)
    {
        ReportMessage(GetLocalI18NString(407));
    }
}

function CheckSave()
{
    try
    {
        var doWeSave = false;
        var edState;
        edState = CombinedEd.State;

        // ISR49581 - do not attempt autosave from a console station
        // if server is unavailable
        if (!IsDisconnectedCStn())
        {
            // Note that if we don't do this 2 step process
            // the script engine always throws an error, despite the
            // catch statement.
            switch(edState)
            {
            case -1:
            case 0:
            case 2:
                // No need to attempt a save
                break;
            case 1:
            case 3:
                doWeSave = true;
                break;
            default:
                // No need to attempt a save
                break;
            }
        }
        if(doWeSave)
        {
            try
            {
                // Don't worry about security here
                CombinedEd.SaveWorking();
            }
            catch(e)
            {
                ReportMessage(GetLocalI18NString(408));
            }
        }
    }
    catch(e)
    {
        // Ignore errors - they probably mean editor was not allowed to run because of security
    }
}

function ReportSetViewErrors(e)
{
    try
    {
        switch(~e.number)
        {
        case ~0x80041503:
            // Hsccli general
            ReportMessage(GetLocalI18NString(412));
            break;
        case ~0x80041504:
            // Point Name Unknown
            ReportMessage(GetLocalI18NString(413));
            break;
        case ~0x80041505:
            // Invalid Point Type
            ReportMessage(GetLocalI18NString(414));
            break;
        case ~0x80041506:
            // Timeout
            ReportMessage(GetLocalI18NString(415));
            break;
        case ~0x80041507:
            // Connect Failure
            ReportMessage(GetLocalI18NString(416));
            break;
        default:
            // Ignore errors - they probably mean editor was not allowed to run because of security
            ReportMessage(GetLocalI18NString(417));
            break;
        }
    }
    catch(e2)
    {
        // Do nothing
    }
}

function AppStatusUpdate()
{

    try
    {
        var blnIsConsole;
        try
        {
            blnIsConsole = window.external.status.IsConsoleStation;
        }
        catch(e)
        {
            blnIsConsole = false;
        }

        if (true == blnIsConsole)
        {

            CurrentState = window.external.status.FieldState(5);

            if (LastState != CurrentState)
            {
                if (CurrentState == nServerUnavailable)
                {
                    if (EditorType == nCombinedScriptEditor)
                    {
                        if (cboScriptType.selectedIndex > -1)
                        {
                            PreviousSelection = cboScriptType.selectedIndex;
                        }
                    }

                    ReportMessage(GetLocalI18NString(419));
                    SelectEditorType();

                }
                else
                {
                    if (LastState == nServerUnavailable)
                    {
                        if (EditorType == nCombinedScriptEditor)
                        {
                            cboScriptType.selectedIndex = PreviousSelection;
                        }
                        cmdSave.style.visibility       = "visible";
                        cmdUndo.style.visibility       = "visible";
                        cmdCommit.style.visibility     = "visible";
                        cmdReset.style.visibility      = "visible";
                        txtCannotSave.style.visibility = "hidden";
                        SelectEditorType();
                    }
                }
                LastState = CurrentState;
            }
        }
    }
    catch(e)
    {
        ReportMessage(GetLocalI18NString(420));
    }
}
